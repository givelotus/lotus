+void V2PeerProtocolHandler::RemovePendingShortTxRequest(
+    NodeId peer_id, ReconScope scope, const uint256 &context_id,
+    uint64_t short_id) {
+    LOCK(m_mutex);
+    const auto it = m_pending_shorttx_requests.find(
+        PendingShortTxKey{peer_id, scope, context_id});
+    if (it == m_pending_shorttx_requests.end()) {
+        return;
+    }
+    it->second.requested_short_ids.erase(short_id);
+    if (it->second.requested_short_ids.empty()) {
+        m_pending_shorttx_requests.erase(it);
+    }
+}

+void V2PeerProtocolHandler::RemovePendingShortTxRequests(
+    ReconScope scope, const uint256 &context_id, uint64_t short_id) {
+    LOCK(m_mutex);
+    // The table has at most MAX_TRACKED_RECON_SCOPES entries. Do not retain
+    // another peer's slot after the ordinary tracker forgets its announcement.
+    for (auto it = m_pending_shorttx_requests.begin();
+         it != m_pending_shorttx_requests.end();) {
+        if (it->first.scope == scope && it->first.context_id == context_id) {
+            it->second.requested_short_ids.erase(short_id);
+            if (it->second.requested_short_ids.empty()) {
+                it = m_pending_shorttx_requests.erase(it);
+                continue;
+            }
+        }
+        ++it;
+    }
+}
